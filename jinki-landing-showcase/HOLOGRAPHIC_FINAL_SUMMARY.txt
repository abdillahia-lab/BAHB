================================================================================
                         CUSTOMIZATION EXAMPLES
================================================================================

CHANGE PRIMARY COLOR:
  :root {
    --holo-cyan: #2dd4bf;        /* Your color */
    --holo-cyan-bright: #5eead4; /* Bright variant */
  }

SLOWER ANIMATIONS:
  @keyframes holo-text-shift {
    animation: holo-text-shift 6s ease-in-out infinite;
  }

DISABLE VHS NOISE ON MOBILE:
  const isMobile = window.innerWidth < 768
  {!isMobile && <VHSNoise intensity={0.05} />}

MORE INTENSE EFFECTS:
  <HolographicCard style={{ '--intensity': 1.5 }} />

CUSTOM BEACON COLOR:
  <HolographicBeacon color="#ff006e" size={80} />
